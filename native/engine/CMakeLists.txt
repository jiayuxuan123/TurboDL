cmake_minimum_required(VERSION 3.24)
project(turbodl_engine VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
if(ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set(TD_DEFAULT_BUILD_CLI OFF)
else()
  set(TD_DEFAULT_BUILD_CLI ON)
endif()
option(TD_BUILD_CLI "Build the standalone CLI" ${TD_DEFAULT_BUILD_CLI})
set(TD_CURL_ROOT "" CACHE PATH "Verified curl development installation")
option(TD_REQUIRE_FULL_TRANSPORT "Require HTTP/2, HTTP/3, FTP and SFTP in the target curl" OFF)
set(TD_SQLITE_SOURCE "" CACHE PATH "Verified sqlite amalgamation directory")
option(TD_BUILD_JNI "Build the JVM/Android bridge" OFF)
set(TD_JAVA_HOME "" CACHE PATH "JDK used only for JNI headers")
option(TD_ENABLE_BITTORRENT "Build the pinned libtorrent protocol module" OFF)
set(TD_LIBTORRENT_SOURCE "" CACHE PATH "Verified libtorrent source directory")
set(TD_BOOST_ROOT "" CACHE PATH "Verified Boost headers")
set(TD_CRYPTO_ROOT "" CACHE PATH "Crypto libraries built with the target compiler")
set(TD_PUGIXML_SOURCE "" CACHE PATH "Verified pugixml source directory")
option(TD_ENABLE_MEDIA "Enable shared-core HLS/DASH and packet muxing" OFF)
set(TD_FFMPEG_ROOT "" CACHE PATH "Pinned FFmpeg libraries, built with networking disabled")
if(NOT EXISTS "${TD_SQLITE_SOURCE}/sqlite3.c")
  message(FATAL_ERROR "Set TD_SQLITE_SOURCE; configuration never downloads dependencies")
endif()
add_library(td_sqlite OBJECT "${TD_SQLITE_SOURCE}/sqlite3.c")
target_include_directories(td_sqlite PUBLIC "${TD_SQLITE_SOURCE}")
target_compile_definitions(td_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_DQS=0 SQLITE_OMIT_LOAD_EXTENSION=1)
if(WIN32 AND TD_CURL_ROOT)
  add_library(td_curl SHARED IMPORTED)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR CMAKE_SYSTEM_PROCESSOR MATCHES "^(ARM64|arm64|aarch64)$")
    message(FATAL_ERROR "The imported libcurl-x64.dll requires a Windows x64 build")
  endif()
  set_target_properties(td_curl PROPERTIES IMPORTED_IMPLIB "${TD_CURL_ROOT}/lib/libcurl.dll.a" IMPORTED_LOCATION "${TD_CURL_ROOT}/bin/libcurl-x64.dll" INTERFACE_INCLUDE_DIRECTORIES "${TD_CURL_ROOT}/include")
  install(FILES "$<TARGET_FILE:td_curl>" DESTINATION bin)
else()
  if(TD_CURL_ROOT)
    set(CURL_ROOT "${TD_CURL_ROOT}")
  endif()
  if(TD_REQUIRE_FULL_TRANSPORT)
    set(OPENSSL_ROOT_DIR "${TD_CRYPTO_ROOT}")
    find_package(CURL 8.2 CONFIG REQUIRED COMPONENTS HTTP2 HTTP3 HTTPS FTP SFTP)
  else()
    find_package(CURL 8.2 REQUIRED)
  endif()
  add_library(td_curl INTERFACE)
  target_link_libraries(td_curl INTERFACE CURL::libcurl)
endif()
find_package(Threads REQUIRED)
add_library(td_hls_manifest OBJECT src/hls_manifest.cpp)
target_link_libraries(td_hls_manifest PRIVATE td_curl)
add_library(turbodl_engine SHARED src/reactor.cpp src/storage.cpp src/resources.cpp src/capabilities.cpp)
target_link_libraries(turbodl_engine PRIVATE td_hls_manifest)
if(TD_PUGIXML_SOURCE)
  if(NOT EXISTS "${TD_PUGIXML_SOURCE}/src/pugixml.cpp")
    message(FATAL_ERROR "TD_PUGIXML_SOURCE must contain the verified source")
  endif()
  target_sources(turbodl_engine PRIVATE src/metalink.cpp "${TD_PUGIXML_SOURCE}/src/pugixml.cpp")
  target_include_directories(turbodl_engine PRIVATE "${TD_PUGIXML_SOURCE}/src")
  target_compile_definitions(turbodl_engine PRIVATE TD_WITH_METALINK=1)
endif()
if(TD_ENABLE_MEDIA)
  if(NOT TD_PUGIXML_SOURCE OR NOT EXISTS "${TD_FFMPEG_ROOT}/include/libavformat/avformat.h")
    message(FATAL_ERROR "Media support requires pinned pugixml and FFmpeg installations")
  endif()
  foreach(component avformat avcodec avutil)
    if(WIN32)
      add_library(td_${component} SHARED IMPORTED)
      if(component STREQUAL "avutil")
        set(major 61)
      else()
        set(major 63)
      endif()
      set_target_properties(td_${component} PROPERTIES IMPORTED_IMPLIB "${TD_FFMPEG_ROOT}/lib/lib${component}.dll.a" IMPORTED_LOCATION "${TD_FFMPEG_ROOT}/bin/${component}-${major}.dll" INTERFACE_INCLUDE_DIRECTORIES "${TD_FFMPEG_ROOT}/include")
      add_custom_command(TARGET turbodl_engine POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:td_${component}>" "$<TARGET_FILE_DIR:turbodl_engine>")
      install(FILES "$<TARGET_FILE:td_${component}>" DESTINATION bin)
    else()
      find_library(TD_${component}_LIBRARY NAMES ${component} PATHS "${TD_FFMPEG_ROOT}/lib"
        REQUIRED NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
      add_library(td_${component} UNKNOWN IMPORTED)
      set_target_properties(td_${component} PROPERTIES IMPORTED_LOCATION "${TD_${component}_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${TD_FFMPEG_ROOT}/include")
    endif()
    target_link_libraries(turbodl_engine PRIVATE td_${component})
  endforeach()
  if(WIN32)
    if(NOT EXISTS "${TD_CRYPTO_ROOT}/lib/libcrypto.a")
      message(FATAL_ERROR "Media AES requires the pinned target crypto build")
    endif()
    target_link_libraries(turbodl_engine PRIVATE "${TD_CRYPTO_ROOT}/lib/libcrypto.a")
    target_include_directories(turbodl_engine PRIVATE "${TD_CRYPTO_ROOT}/include")
  endif()
  target_sources(turbodl_engine PRIVATE src/media.cpp src/media_mux.cpp src/media_state.cpp src/dash_manifest.cpp)
  target_compile_definitions(turbodl_engine PRIVATE TD_WITH_MEDIA=1)
  if(BUILD_TESTING)
    add_executable(td-media-fixture tests/media_fixture.cpp)
    target_link_libraries(td-media-fixture PRIVATE td_avformat td_avcodec td_avutil)
  endif()
endif()
target_include_directories(turbodl_engine PUBLIC include)
target_compile_definitions(turbodl_engine PRIVATE TD_ENGINE_BUILD)
if(TD_ENABLE_BITTORRENT)
  if(NOT EXISTS "${TD_LIBTORRENT_SOURCE}/CMakeLists.txt" OR NOT EXISTS "${TD_BOOST_ROOT}/boost/version.hpp")
    message(FATAL_ERROR "BitTorrent requires verified libtorrent sources and Boost headers")
  endif()
  set(CMAKE_POLICY_DEFAULT_CMP0167 OLD)
  set(BOOST_ROOT "${TD_BOOST_ROOT}")
  set(Boost_INCLUDE_DIR "${TD_BOOST_ROOT}" CACHE PATH "" FORCE)
  set(Boost_NO_BOOST_CMAKE ON CACHE BOOL "" FORCE)
  set(Boost_NO_SYSTEM_PATHS ON CACHE BOOL "" FORCE)
  if(WIN32 AND TD_CURL_ROOT)
    if(NOT TD_CRYPTO_ROOT)
      message(FATAL_ERROR "Build target crypto dependencies and set TD_CRYPTO_ROOT")
    endif()
    set(OPENSSL_ROOT_DIR "${TD_CRYPTO_ROOT}" CACHE PATH "" FORCE)
    set(OPENSSL_INCLUDE_DIR "${TD_CRYPTO_ROOT}/include" CACHE PATH "" FORCE)
    set(OPENSSL_SSL_LIBRARY "${TD_CRYPTO_ROOT}/lib/libssl.a" CACHE FILEPATH "" FORCE)
    set(OPENSSL_CRYPTO_LIBRARY "${TD_CRYPTO_ROOT}/lib/libcrypto.a" CACHE FILEPATH "" FORCE)
    set(OPENSSL_SSL_LIBRARY_RELEASE "${TD_CRYPTO_ROOT}/lib/libssl.a" CACHE FILEPATH "" FORCE)
    set(OPENSSL_SSL_LIBRARY_DEBUG "${TD_CRYPTO_ROOT}/lib/libssl.a" CACHE FILEPATH "" FORCE)
    set(OPENSSL_CRYPTO_LIBRARY_RELEASE "${TD_CRYPTO_ROOT}/lib/libcrypto.a" CACHE FILEPATH "" FORCE)
    set(OPENSSL_CRYPTO_LIBRARY_DEBUG "${TD_CRYPTO_ROOT}/lib/libcrypto.a" CACHE FILEPATH "" FORCE)
    set(LIB_EAY "${TD_CRYPTO_ROOT}/lib/libcrypto.a" CACHE FILEPATH "" FORCE)
    set(SSL_EAY "${TD_CRYPTO_ROOT}/lib/libssl.a" CACHE FILEPATH "" FORCE)
    set(OPENSSL_USE_STATIC_LIBS ON CACHE BOOL "" FORCE)
  endif()
  set(build_tests OFF CACHE BOOL "" FORCE)
  set(build_examples OFF CACHE BOOL "" FORCE)
  set(python-bindings OFF CACHE BOOL "" FORCE)
  set(webtorrent OFF CACHE BOOL "" FORCE)
  if(ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
    set(BUILD_SHARED_LIBS OFF)
  else()
    set(BUILD_SHARED_LIBS ON)
  endif()
  add_subdirectory("${TD_LIBTORRENT_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/libtorrent" EXCLUDE_FROM_ALL)
  target_link_libraries(turbodl_engine PRIVATE torrent-rasterbar)
  target_sources(turbodl_engine PRIVATE src/torrent.cpp)
  target_compile_definitions(turbodl_engine PRIVATE TD_WITH_BITTORRENT=1)
  if(BUILD_SHARED_LIBS)
    add_custom_command(TARGET turbodl_engine POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:torrent-rasterbar>" "$<TARGET_FILE_DIR:turbodl_engine>")
    install(TARGETS torrent-rasterbar RUNTIME DESTINATION bin LIBRARY DESTINATION lib)
  endif()
  if(BUILD_TESTING)
    add_executable(td-torrent-fixture tests/torrent_seed.cpp)
    target_link_libraries(td-torrent-fixture PRIVATE torrent-rasterbar)
    if(BUILD_SHARED_LIBS)
      add_custom_command(TARGET td-torrent-fixture POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:torrent-rasterbar>" "$<TARGET_FILE_DIR:td-torrent-fixture>")
    endif()
  endif()
endif()
if(TD_BUILD_JNI)
  target_sources(turbodl_engine PRIVATE src/jni.cpp)
  if(NOT ANDROID)
    find_path(TD_JNI_INCLUDE jni.h PATHS "${TD_JAVA_HOME}/include" REQUIRED NO_DEFAULT_PATH)
    target_include_directories(turbodl_engine PRIVATE "${TD_JNI_INCLUDE}")
    if(WIN32)
      target_include_directories(turbodl_engine PRIVATE "${TD_JNI_INCLUDE}/win32")
    elseif(APPLE)
      target_include_directories(turbodl_engine PRIVATE "${TD_JNI_INCLUDE}/darwin")
    else()
      target_include_directories(turbodl_engine PRIVATE "${TD_JNI_INCLUDE}/linux")
    endif()
  endif()
endif()
target_link_libraries(turbodl_engine PRIVATE td_curl td_sqlite Threads::Threads)
target_compile_options(turbodl_engine PRIVATE -Wall -Wextra -Wpedantic)
if(WIN32)
  target_link_libraries(turbodl_engine PRIVATE bcrypt ws2_32)
  target_compile_definitions(turbodl_engine PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  set_target_properties(turbodl_engine PROPERTIES PREFIX "")
  if(TD_CURL_ROOT)
    add_custom_command(TARGET turbodl_engine POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:td_curl>" "$<TARGET_FILE_DIR:turbodl_engine>")
  endif()
else()
  if(TD_CRYPTO_ROOT)
    set(OPENSSL_ROOT_DIR "${TD_CRYPTO_ROOT}")
  endif()
  find_package(OpenSSL REQUIRED)
  target_link_libraries(turbodl_engine PRIVATE OpenSSL::SSL OpenSSL::Crypto)
endif()
if(TD_BUILD_CLI)
  add_executable(tdl-vnext src/cli.cpp)
  target_link_libraries(tdl-vnext PRIVATE turbodl_engine)
  if(WIN32)
    target_link_libraries(tdl-vnext PRIVATE shell32)
  elseif(APPLE)
    set_target_properties(tdl-vnext PROPERTIES INSTALL_RPATH "@loader_path/../lib")
    set_target_properties(turbodl_engine PROPERTIES INSTALL_RPATH "@loader_path")
  else()
    set_target_properties(tdl-vnext PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
    set_target_properties(turbodl_engine PROPERTIES INSTALL_RPATH "$ORIGIN")
  endif()
  install(TARGETS tdl-vnext RUNTIME DESTINATION bin)
endif()
if(BUILD_TESTING)
  if(WIN32 AND TD_ENABLE_MEDIA)
    add_executable(td-hash-bench tests/hash_bench.cpp)
    target_include_directories(td-hash-bench PRIVATE "${TD_CRYPTO_ROOT}/include")
    target_link_libraries(td-hash-bench PRIVATE td_avutil "${TD_CRYPTO_ROOT}/lib/libcrypto.a" bcrypt ws2_32)
  endif()
  add_executable(td-transfer-bench tests/transfer_bench.cpp)
  target_link_libraries(td-transfer-bench PRIVATE turbodl_engine)
  add_executable(td-hls-manifest-tests tests/hls_manifest_test.cpp)
  target_link_libraries(td-hls-manifest-tests PRIVATE td_hls_manifest td_curl)
  target_include_directories(td-hls-manifest-tests PRIVATE src)
  add_executable(td-ed2k-p0-tests tests/ed2k_p0_test.cpp src/ed2k/ed2k_link.cpp src/ed2k/ed2k_hash.cpp)
  target_include_directories(td-ed2k-p0-tests PRIVATE src)
  if(WIN32)
    target_link_libraries(td-ed2k-p0-tests PRIVATE ws2_32)
  endif()
  add_test(NAME hls_manifest_parser COMMAND td-hls-manifest-tests)
  add_test(NAME ed2k_p0 COMMAND td-ed2k-p0-tests)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME native_c_abi COMMAND ${CMAKE_COMMAND} -E env "TD_ENGINE_LIBRARY=$<TARGET_FILE:turbodl_engine>" ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_engine.py")
  set_tests_properties(native_c_abi PROPERTIES TIMEOUT 90)
endif()
install(TARGETS turbodl_engine LIBRARY DESTINATION lib RUNTIME DESTINATION bin ARCHIVE DESTINATION lib)
install(FILES include/turbodl/engine.h DESTINATION include/turbodl)
